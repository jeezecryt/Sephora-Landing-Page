function toggle(btn) {
  var item = btn.parentElement;

  var mainCta = document.getElementById('mainCta');
  var sticky = document.getElementById('stickyCta');
  var obs = new IntersectionObserver(function(entries){
    entries.forEach(function(e){

  }, { threshold: 0 });
  obs.observe(mainCta);
})();
